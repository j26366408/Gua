指揮 AI 的作業內容

建立可讀取整數 N、使用迴圈計算 1＋2＋⋯＋N 的 Fortran 程式，編譯後實際執行。

使用 integer(int64) 儲存總和；本程式接受 1 ≤ N ≤ 10,000,000，並檢查讀取失敗與超出範圍的輸入。

編譯及範例輸出

gfortran -std=f2008 -Wall -Wextra -Werror \
  -fcheck=all -O2 src/sum_to_n.f90 -o build/sum_to_n
echo 10 | ./build/sum_to_n

Enter N (1..10000000):
N = 10, SUM = 55

GNU Fortran 14.2.0；原版編譯結束碼 0，沒有警告或錯誤。執行環境採解壓縮於工作目錄的 GNU 工具鏈，完整參數附於紀錄。

sum_to_n.f90 · 逐行註解版

program sum_to_n ! 宣告主程式名稱
  use iso_fortran_env, only: int64, error_unit ! 使用 64 位元整數與標準錯誤輸出
  implicit none ! 禁止未宣告變數
  integer(int64) :: n, i, total ! 宣告上限、迴圈變數與加總結果
  integer :: ios ! 宣告輸入狀態碼
  print '(A)', 'Enter N (1..10000000):' ! 提示可接受的整數範圍
  read (*, *, iostat=ios) n ! 讀入 N 並記錄讀取狀態
  if (ios /= 0) then ! 判斷輸入是否失敗
    write (error_unit, '(A)') 'ERROR: input must be an integer.' ! 輸出非整數輸入的錯誤訊息
    stop 1 ! 以非零狀態結束程式
  end if ! 結束輸入失敗判斷
  if (n < 1_int64 .or. n > 10000000_int64) then ! 檢查 N 是否超出範圍
    write (error_unit, '(A)') 'ERROR: N must be in 1..10000000.' ! 輸出範圍錯誤訊息
    stop 1 ! 以非零狀態結束程式
  end if ! 結束範圍檢查
  total = 0_int64 ! 將累加結果初始化為零
  do i = 1_int64, n ! 從 1 迴圈到 N，預設每次加 1
    total = total + i ! 把本次整數加入總和
  end do ! 結束加總迴圈
  print '(A,I0,A,I0)', 'N = ', n, ', SUM = ', total ! 印出 N 與總和，I0 不保留多餘空白
end program sum_to_n ! 結束主程式

查看驗證腳本 verify.py
"""Run the real Fortran executable, compare with an independent arithmetic oracle."""
import json, re, secrets, subprocess
from pathlib import Path
root=Path(__file__).resolve().parent
rng=secrets.SystemRandom()
ns=rng.sample(range(1,100001),3)
rows=[]
for n in ns:
 p=subprocess.run([str(root/'build/sum_to_n')],input=f'{n}\n',text=True,capture_output=True)
 match=re.search(r'SUM = (\d+)',p.stdout)
 actual=int(match.group(1)) if match else None
 expected=n*(n+1)//2
 rows.append(dict(n=n,expected=expected,actual=actual,returncode=p.returncode,stdout=p.stdout,stderr=p.stderr,passed=p.returncode==0 and actual==expected))
report={'method':'secrets.SystemRandom.sample(range(1,100001),3); no fixed seed; independent N*(N+1)//2 oracle','tests':rows,'all_passed':all(r['passed'] for r in rows)}
(root/'evidence/random-tests.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
for r in rows: print(f"N={r['n']} actual={r['actual']} expected={r['expected']} {'PASS' if r['passed'] else 'FAIL'}")
raise SystemExit(0 if report['all_passed'] else 1)
